                    params={"symbol": symbol.upper(), "timeframe": timeframe, "page_size": 100},
                    timeout=30.0
                )

                if not candles_data:

                        symbol=symbol.upper(), timeframe=timeframe, timestamp=datetime.now(timezone.utc),

                        reason="No data available"
